配置文件如下：
[profile.default]
src = "src"
test = "test"
script = "script"
out = "out"
libs = ["dependencies"]  # 关键配置：指定依赖目录

# 编译器配置
solc = "0.8.20"
optimizer = true
optimizer_runs = 200
remappings = ["@forge-std-1.16.0/=dependencies/forge-std-1.16.0/"]

# 依赖配置 - 仅需包名和版本
# 项目依赖列表
[dependencies]
"@openzeppelin-contracts" = { version = "5.0.2" }
forge-std = "1.16.0"


# Soldeer 的全局配置
[soldeer]
remappings_generate = true
remappings_location = "config" # 重映射会直接写入这个 foundry.toml 文件中
remappings_prefix = "@"
recursive_deps = false



# 路径映射配置
remappings = [
    "forge-std/=dependencies/forge-std/src/",
    "@openzeppelin/contracts/=dependencies/openzeppelin-contracts/contracts/",
    "@openzeppelin/=dependencies/openzeppelin-contracts/",
    "@uniswap/v4-core/=dependencies/uniswap-v4-core/",
    "src/=src/",
    "test/=test/",
    "script/=script/"
]
